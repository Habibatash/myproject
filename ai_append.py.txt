def _agency_extract_attachment(att):
    """Extract readable text only; never claim to understand binary media."""
    try:
        if not att.file:
            return ""
        path = att.file.path
        if not _agency_os.path.isfile(path) or _agency_os.path.getsize(path) > 4 * 1024 * 1024:
            return ""
        ext = _AgencyPath(att.name or path).suffix.lower()
        raw = open(path, "rb").read(4 * 1024 * 1024 + 1)
        if ext in (".txt", ".md", ".csv", ".json", ".xml", ".log"):
            return raw.decode("utf-8", errors="replace")[:12000]
        if ext == ".docx":
            from docx import Document as _Docx
            doc = _Docx(_agency_io.BytesIO(raw))
            parts = [p.text for p in doc.paragraphs if p.text]
            for table in doc.tables:
                for row in table.rows:
                    parts.append(" | ".join(c.text for c in row.cells))
            return "\n".join(parts)[:12000]
        if ext == ".xlsx":
            from openpyxl import load_workbook as _load_workbook
            wb = _load_workbook(_agency_io.BytesIO(raw), read_only=True, data_only=True)
            out = []
            for ws in wb.worksheets[:10]:
                out.append("[Sheet: " + ws.title + "]")
                for row in ws.iter_rows(max_row=200, max_col=30, values_only=True):
                    vals = [str(v) for v in row if v is not None]
                    if vals:
                        out.append(" | ".join(vals))
            return "\n".join(out)[:12000]
        if ext == ".pdf":
            try:
                from pypdf import PdfReader as _PdfReader
            except ImportError:
                from PyPDF2 import PdfReader as _PdfReader
            reader = _PdfReader(_agency_io.BytesIO(raw))
            return "\n".join((page.extract_text() or "") for page in reader.pages[:40])[:12000]
    except Exception as exc:  # noqa: BLE001
        _agency_log.warning("AI could not extract attachment %s: %s", getattr(att, "name", "?"), exc)
    return ""


def collect_items(me, days):
    since = timezone.now() - timedelta(days=days)
    dnames = {d.pk: d.name for d in Department.objects.all()}
    items = []
    # Public internal news is visible to every authenticated user.
    for p in Post.objects.filter(created__gte=since).select_related("author", "department").prefetch_related("comments", "attachments").order_by("-created")[:MAX_ITEMS]:
        text = p.text or ""
        for att in p.attachments.all()[:8]:
            ext_text = _agency_extract_attachment(att)
            if ext_text:
                text += "\n[Attachment " + att.name + "]\n" + ext_text
        items.append({"type": "news-important" if p.kind == "important" else "news-general", "date": p.created.strftime("%Y-%m-%d"), "dept": dnames.get(p.department_id, "all"), "by": p.author.name if p.author else p.author_name, "text": text, "comments": p.comments.count()})
    for d in visible_documents(me).filter(created__gte=since).select_related("owner", "department").prefetch_related("attachments").order_by("-created")[:MAX_ITEMS]:
        text = f"{d.title}. {d.description}".strip()
        for att in d.attachments.all()[:8]:
            ext_text = _agency_extract_attachment(att)
            if ext_text:
                text += "\n[Attachment " + att.name + "]\n" + ext_text
        items.append({"type": "document", "date": d.created.strftime("%Y-%m-%d"), "dept": dnames.get(d.department_id, ""), "by": d.employee or (d.owner.name if d.owner else ""), "text": text, "period": d.period, "files": [a.name for a in d.attachments.all()][:8]})
    items.sort(key=lambda x: x["date"], reverse=True)
    return items


def answer(me, question, days):
    if not ai_enabled():
        raise RuntimeError("AI_NOT_CONFIGURED")

    result = ask_claude(question, items)
